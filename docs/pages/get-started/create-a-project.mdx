<script>
    // 상태 배열 정의
    const statusCycle = [
        { text: '대기', class: 'status-wait' },
        { text: '완료(O)', class: 'status-done' },
        { text: '미완료(X)', class: 'status-fail' },
        { text: '이월(→)', class: 'status-move' }
    ];

    let todos = [];
    let timerInterval = null;
    let secondsSpent = 0;
    let totalSeconds = 0;
    let isRunning = false;

    // 초기 날짜 세팅 (오늘 날짜)
    const dateInput = document.getElementById('planner-date');
    const today = new Date().toISOString().split('T')[0];
    dateInput.value = today;

    // 날짜 변경 시 이벤트
    dateInput.addEventListener('change', () => {
        loadData(dateInput.value);
    });

    // 할 일 추가 기능
    function addTodo() {
        const todoText = document.getElementById('todo-input').value.trim();
        const csfText = document.getElementById('csf-input').value.trim();

        if (!todoText) {
            alert('할 일을 입력해 주세요!');
            return;
        }

        const newTodo = {
            id: Date.now(),
            text: todoText,
            csf: csfText ? `(${csfText})` : '',
            statusIndex: 0
        };

        todos.push(newTodo);
        renderTodos();

        document.getElementById('todo-input').value = '';
        document.getElementById('csf-input').value = '';
    }

    // 할 일 목록 렌더링
    function renderTodos() {
        const listContainer = document.getElementById('todo-list');
        listContainer.innerHTML = '';

        todos.forEach((todo, index) => {
            const item = document.createElement('div');
            item.className = 'todo-item';

            const currentStatus = statusCycle[todo.statusIndex];

            item.innerHTML = `
                <div class="todo-text">
                    <strong>${todo.text}</strong>
                    ${todo.csf ? `<span class="csf">핵심 요인: ${todo.csf}</span>` : ''}
                </div>
                <div>
                    <button class="status-btn ${currentStatus.class}" onclick="cycleStatus(${index})">
                        ${currentStatus.text}
                    </button>
                    <button onclick="deleteTodo(${index})" style="background:none; color:#ef4444; padding:5px; margin-left:5px;">🗑️</button>
                </div>
            `;
            listContainer.appendChild(item);
        });
    }

    // O, X, 화살표 상태 전환 기능
    function cycleStatus(index) {
        todos[index].statusIndex = (todos[index].statusIndex + 1) % statusCycle.length;
        renderTodos();
    }

    // 할 일 삭제 기능
    function deleteTodo(index) {
        todos.splice(index, 1);
        renderTodos();
    }

    // 타이머 기능 제어
    function toggleTimer() {
        const btn = document.getElementById('btn-start');
        if (!isRunning) {
            isRunning = true;
            btn.textContent = '일시정지';
            btn.style.backgroundColor = '#f59e0b';
            timerInterval = setInterval(() => {
                secondsSpent++;
                document.getElementById('timer-view').textContent = formatTime(secondsSpent);
            }, 1000);
        } else {
            isRunning = false;
            btn.textContent = '시작';
            btn.style.backgroundColor = '#10b981';
            clearInterval(timerInterval);
            totalSeconds += secondsSpent;
            secondsSpent = 0;
            document.getElementById('timer-view').textContent = '00:00:00';
            document.getElementById('total-time-view').textContent = formatTime(totalSeconds);
        }
    }

    function resetTimer() {
        clearInterval(timerInterval);
        isRunning = false;
        secondsSpent = 0;
        totalSeconds = 0;
        document.getElementById('btn-start').textContent = '시작';
        document.getElementById('btn-start').style.backgroundColor = '#10b981';
        document.getElementById('timer-view').textContent = '00:00:00';
        document.getElementById('total-time-view').textContent = '00:00:00';
    }

    function formatTime(sec) {
        const hrs = String(Math.floor(sec / 3600)).padStart(2, '0');
        const mins = String(Math.floor((sec % 3600) / 60)).padStart(2, '0');
        const secs = String(sec % 60).padStart(2, '0');
        return `${hrs}:${mins}:${secs}`;
    }

    // 브라우저 로컬 저장소(LocalStorage)에 데이터 저장
    function saveAllData() {
        const dateKey = dateInput.value;
        
        // 만약 타이머가 켜져있는 상태로 저장하면 현재 흘러간 시간까지 합산
        if (isRunning) {
            totalSeconds += secondsSpent;
            secondsSpent = 0;
            clearInterval(timerInterval);
            isRunning = false;
            document.getElementById('btn-start').textContent = '시작';
            document.getElementById('btn-start').style.backgroundColor = '#10b981';
            document.getElementById('timer-view').textContent = '00:00:00';
            document.getElementById('total-time-view').textContent = formatTime(totalSeconds);
        }

        const dayData = {
            todos: todos,
            totalSeconds: totalSeconds,
            gratitude: document.getElementById('diary-gratitude').value,
            relationship: document.getElementById('diary-relationship').value,
            review: document.getElementById('diary-review').value
        };

        localStorage.setItem(`daegun_planner_${dateKey}`, JSON.stringify(dayData));
        alert(`${dateKey} 플래너와 일기가 성공적으로 저장되었습니다!`);
    }

    // 데이터 불러오기 기능
    function loadData(dateKey) {
        const saved = localStorage.getItem(`daegun_planner_${dateKey}`);
        if (saved) {
            const data = JSON.parse(saved);
            todos = data.todos || [];
            totalSeconds = data.totalSeconds || 0;
            document.getElementById('diary-gratitude').value = data.gratitude || '';
            document.getElementById('diary-relationship').value = data.relationship || '';
            document.getElementById('diary-review').value = data.review || '';
        } else {
            // 저장된 데이터가 없는 새 날짜일 때 초기화
            todos = [];
            totalSeconds = 0;
            document.getElementById('diary-gratitude').value = '';
            document.getElementById('diary-relationship').value = '';
            document.getElementById('diary-review').value = '';
        }
        renderTodos();
        document.getElementById('total-time-view').textContent = formatTime(totalSeconds);
        document.getElementById('timer-view').textContent = '00:00:00';
    }

    // 페이지 첫 로드시 오늘 데이터 자동 로드
    loadData(today);
</script>
